    HDRS
      cjk_tables.h
      code_table.h
    SRCS
      cjk_tables.cpp
    DEPENDS
      libc.hdr.stdint_proxy
      libc.hdr.types.char32_t

      libc.src.__support.macros.attributes
      libc.src.__support.macros.config
